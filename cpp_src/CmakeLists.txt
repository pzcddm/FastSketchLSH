cmake_minimum_required(VERSION 3.16)  # Raise minimum version to support modern features
project(FastSketchLSH LANGUAGES CXX)

# Cross-platform compile options
if(MSVC)
    add_compile_options(/std:c++17 /fp:fast)
    add_definitions(-D_USE_MATH_DEFINES -DNOMINMAX)
else()
    # Architecture-safe defaults; avoid x86-only flags on ARM/others
    add_compile_options(-std=c++17 -ffast-math)
    set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
endif()

# Dynamic detection of SIMD instruction sets
option(ENABLE_NEON "Enable ARM NEON instructions (auto on arm64)" ON)
option(ENABLE_AVX512 "Enable AVX-512 instructions (x86_64 only)" ON)
option(FASTSKETCH_USE_FXHASH "Use FxHasher-style prehash for strings" ON)

# Auto-disable AVX512 on non-x86 and hint NEON on arm64
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|i.86")
    set(ENABLE_AVX512 OFF)
endif()
if(FASTSKETCH_USE_FXHASH)
    add_definitions(-DFASTSKETCH_USE_FXHASH)
endif()

# PyBind11 integration (optional, for Python module)
find_package(pybind11 QUIET)

# Core library target (core sources, excluding a separate AVX-512 TU)
set(FASTSKETCH_CORE_SRCS
    cpp/rminhash.cpp
    # fasthash scalar implementation is deprecated; keep source for archival only
    # cpp/fasthash.cpp
    cpp/fasthash_deprecated.cpp
    cpp/fastsketch.cpp
    cpp/murmurhash3.cpp
    cpp/fastsketch_lsh.cpp
    cpp/fastsketch_rensa_lsh.cpp
)

add_library(FastSketchLSH_core STATIC ${FASTSKETCH_CORE_SRCS})
target_include_directories(FastSketchLSH_core PUBLIC include)

# Add AVX-512 source on demand and set per-file compile options
if(ENABLE_AVX512)
    target_sources(FastSketchLSH_core PRIVATE cpp/fastsketch.cpp)
    if(MSVC)
        # Enable /arch:AVX512 only for fastsketch.cpp
        set_source_files_properties(cpp/fastsketch.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX512")
    else()
        # Enable AVX-512 instruction set only for this TU
        set_source_files_properties(cpp/fastsketch.cpp PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512dq;-mavx512vl")
    endif()
else()
    message(WARNING "ENABLE_AVX512=OFF; skipping cpp/fastsketch.cpp (SIMD sketch will be unavailable)")
endif()

# Python module (build if pybind11 is found, otherwise skip)
if(pybind11_FOUND)
    pybind11_add_module(FastSketchLSH cpp/init.cpp)
    target_link_libraries(FastSketchLSH PRIVATE FastSketchLSH_core pybind11::module)
else()
    message(WARNING "pybind11 not found; skipping Python module build (FastSketchLSH)")
endif()

# Install configuration
if(TARGET FastSketchLSH)
    install(TARGETS FastSketchLSH LIBRARY DESTINATION ${CMAKE_INSTALL_PREFIX})
endif()

# --------------------- Tests / Benchmarks ---------------------
# Standalone performance comparison executable
add_executable(test_rminhash_vs_fasthash_performance
    test/test_rminhash_vs_fasthash_performance.cpp)
target_include_directories(test_rminhash_vs_fasthash_performance PRIVATE include)
target_link_libraries(test_rminhash_vs_fasthash_performance PRIVATE FastSketchLSH_core)
