cmake_minimum_required(VERSION 3.12)
project(FastSketchLSH LANGUAGES CXX)

# 设置编译选项
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -march=native")

# 查找PyBind11
find_package(pybind11 REQUIRED)

# 添加库目标
add_library(FastSketchLSH
    cpp/cminhash.cpp
    cpp/kminhash.cpp
    cpp/rminhash.cpp
    cpp/murmurhash3.cpp
)
target_include_directories(FastSketchLSH PUBLIC include)

# Python模块
pybind11_add_module(FastSketchLSH cpp/init.cpp)
target_link_libraries(FastSketchLSH PRIVATE FastSketchLSH)
